cmake_minimum_required(VERSION 3.16)
project(sim C)
set(CMAKE_C_STANDARD 11)
set(IDF_PATH $ENV{IDF_PATH})
if(NOT IDF_PATH)
    set(IDF_PATH $ENV{HOME}/esp/esp-idf)
endif()
set(CJSON_DIR ${IDF_PATH}/components/json/cJSON)
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

set(LV_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(${ROOT}/managed_components/lvgl__lvgl lvgl)
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_include_directories(lvgl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

add_subdirectory(${ROOT}/components/core core)
add_subdirectory(${ROOT}/components/ui ui)
target_compile_options(ui PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -Wno-missing-field-initializers)
target_compile_options(core PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)

add_executable(sim main.c)
target_compile_options(sim PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter -Wno-missing-field-initializers)
target_compile_definitions(sim PRIVATE FIXTURES="${ROOT}/fixtures")
target_link_libraries(sim ui core lvgl)
