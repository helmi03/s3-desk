cmake_minimum_required(VERSION 3.16)
project(core_tests C)
set(CMAKE_C_STANDARD 11)
set(IDF_PATH $ENV{IDF_PATH})
if(NOT IDF_PATH)
    set(IDF_PATH $ENV{HOME}/esp/esp-idf)
endif()
set(CJSON_DIR ${IDF_PATH}/components/json/cJSON)
set(UNITY_DIR ${IDF_PATH}/components/unity/unity/src)
add_compile_options(-Wall -Wextra -Werror -Wno-unused-parameter -fsanitize=address,undefined -g)
add_link_options(-fsanitize=address,undefined)
add_subdirectory(../components/core core)
add_executable(core_tests test_main.c ${UNITY_DIR}/unity.c)
target_include_directories(core_tests PRIVATE ${UNITY_DIR})
target_compile_definitions(core_tests PRIVATE FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/../fixtures")
target_link_libraries(core_tests core)
